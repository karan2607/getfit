# Python >= 3.10 (the `anthropic` SDK's floor). Production runs 3.14 and CI
# covers 3.12/3.13/3.14.
#
# Django has an UPPER bound on purpose. It was `django>=4.2` with none, and
# production runs pip install on every deploy, so the framework crossed a
# major version with no commit and no diff -- which is how a 4.2-green build
# met Django 5 in production and 500'd the health page. <7 keeps security
# patches flowing and makes the next major a reviewable commit.
# Currently resolves to 6.1.2. See api/tests/test_runtime_compat.py.
django>=4.2,<7
djangorestframework
django-cors-headers
# Claude Haiku 5.5 drives the workout half of the chat agent. Its floor
# is Python 3.10, which is why 3.9 left the CI matrix.
anthropic>=2.3,<3
psycopg2-binary
python-dotenv
gunicorn
whitenoise
sentry-sdk[django]
